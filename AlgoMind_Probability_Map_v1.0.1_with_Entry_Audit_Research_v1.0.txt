add(float p,int side,int rank,string source)=>

    if not na(p)

        int found=-1

        if array.size(levels)>0

            for i=0 to array.size(levels)-1

                if array.get(live,i) and array.get(sides,i)==side and math.abs(array.get(levels,i)-p)<=math.max(syminfo.mintick*3,atr*.012)

                    found:=i

                    break

        if found>=0

            string old=array.get(sources,found)

            if not str.contains(old,source)

                array.set(sources,found,rank<array.get(ranks,found)?source+" + "+old:old+" + "+source)

            if rank<array.get(ranks,found)

                array.set(ranks,found,rank)

                array.set(levels,found,p)

        else

            array.push(levels,p),array.push(sides,side),array.push(ranks,rank),array.push(sources,source),array.push(live,true)

pm_find_state(string key)=>
    int idx=-1
    if array.size(pmKeys)>0
        for i=0 to array.size(pmKeys)-1
            if array.get(pmKeys,i)==key
                idx:=i
                break
    idx

pm_ensure_state(string key)=>
    int idx=pm_find_state(key)
    if idx<0
        array.push(pmKeys,key)
        array.push(pmNs,0)
        for d=0 to PM_DESTS-1
            array.push(pmHits,0)
        idx:=array.size(pmKeys)-1
    idx

pm_record(string key,int dest)=>
    int idx=pm_ensure_state(key)
    array.set(pmNs,idx,array.get(pmNs,idx)+1)
    if dest>=0
        int hidx=idx*PM_DESTS+dest
        array.set(pmHits,hidx,array.get(pmHits,hidx)+1)

    int n=idx>=0?array.get(pmNs,idx):0
    int h=idx>=0?array.get(pmHits,idx*PM_DESTS+dest):0
    float p=n>0?100.0*h/n:na
    [p,n]
